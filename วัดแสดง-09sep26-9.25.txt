function doGet(e) {
  var action = (e && e.parameter && e.parameter.action) ? e.parameter.action.trim() : "";

  // 1.1 สรุปสถิติรายอาคารสำหรับ PWA Dashboard
  if (action === "getBuildingStats") {
    try {
      var ss = SpreadsheetApp.getActiveSpreadsheet();
      var areaSheet = ss.getSheetByName("Database_Area");
      var summary = {};

      if (areaSheet && areaSheet.getLastRow() > 1) {
        var values = areaSheet.getRange(2, 1, areaSheet.getLastRow() - 1, 20).getDisplayValues();
        var roomMap = {};

        for (var i = 0; i < values.length; i++) {
          var bld = (values[i][8] || "").trim(); // Col I: อาคาร
          var room = (values[i][9] || "").trim(); // Col J: ห้อง
          var ev = (values[i][17] || "").trim(); // Col R: ผลประเมิน
          if (!bld || !room) continue;

          var key = bld + "___" + room;
          if (!roomMap[key]) roomMap[key] = { bld: bld, isPass: true };
          if (ev.indexOf("ไม่ผ่าน") !== -1) roomMap[key].isPass = false;
        }

        for (var rKey in roomMap) {
          var b = roomMap[rKey].bld;
          if (!summary[b]) summary[b] = { passRooms: 0, failRooms: 0, totalRooms: 0 };
          summary[b].totalRooms++;
          if (roomMap[rKey].isPass) summary[b].passRooms++;
          else summary[b].failRooms++;
        }
      }

      return ContentService.createTextOutput(JSON.stringify(summary)).setMimeType(ContentService.MimeType.JSON);
    } catch (err) {
      return ContentService.createTextOutput(JSON.stringify({ error: err.toString() })).setMimeType(ContentService.MimeType.JSON);
    }
  }

  // 1.2 ส่งข้อมูลตรวจวัดแบบพื้นที่ (Database_Area)
  if (action === "getAreaSummary") {
    try {
      var ss = SpreadsheetApp.getActiveSpreadsheet();
      var sheet = ss.getSheetByName("Database_Area");
      if (!sheet || sheet.getLastRow() <= 1) {
        return ContentService.createTextOutput(JSON.stringify([])).setMimeType(ContentService.MimeType.JSON);
      }

      var lastRow = sheet.getLastRow();
      var numRows = Math.min(lastRow - 1, 200);
      var data = sheet.getRange(lastRow - numRows + 1, 1, numRows, 20).getDisplayValues().reverse();

      var result = data.map(function(r) {
        return {
          timestamp: r[0], date: r[1], time: r[2], weather: r[3],
          equipment: r[4], serialNo: r[5], calDate: r[6],
          department: r[7], building: r[8], room: r[9],
          pointCode: r[10], task: r[11], pattern: r[12],
          pointLux: Number(r[13]) || 0,
          standardLux: r[14], minLux: r[15],
          measuredLux: Number(r[16]) || 0,
          evaluation: r[17], recommendation: r[18], inspectorEmail: r[19]
        };
      });

      return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON);
    } catch (err) {
      return ContentService.createTextOutput(JSON.stringify({ error: err.toString() })).setMimeType(ContentService.MimeType.JSON);
    }
  }

  // 1.3 ส่งข้อมูลตรวจวัดเฉพาะจุด (Database_Spot)
  if (action === "getSpotSummary") {
    try {
      var ss = SpreadsheetApp.getActiveSpreadsheet();
      var sheet = ss.getSheetByName("Database_Spot");
      if (!sheet || sheet.getLastRow() <= 1) {
        return ContentService.createTextOutput(JSON.stringify([])).setMimeType(ContentService.MimeType.JSON);
      }

      var lastRow = sheet.getLastRow();
      var numRows = Math.min(lastRow - 1, 200);
      var data = sheet.getRange(lastRow - numRows + 1, 1, numRows, 19).getDisplayValues().reverse();

      var result = data.map(function(r) {
        return {
          timestamp: r[0], date: r[1], time: r[2], weather: r[3],
          equipment: r[4], serialNo: r[5], calDate: r[6],
          department: r[7], building: r[8], room: r[9],
          workerName: r[10], task: r[11],
          luxArea1: Number(r[12]) || 0,
          luxArea2: Number(r[13]) || 0,
          luxArea3: Number(r[14]) || 0,
          standardLux: r[15],
          evaluation: r[16], recommendation: r[17], inspectorEmail: r[18]
        };
      });

      return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON);
    } catch (err) {
      return ContentService.createTextOutput(JSON.stringify({ error: err.toString() })).setMimeType(ContentService.MimeType.JSON);
    }
  }

  return HtmlService.createHtmlOutputFromFile('Index')
    .setTitle('ระบบบันทึกผลการตรวจวัดแสงสว่าง ISO 45001')
    .addMetaTag('viewport', 'width=device-width, initial-scale=1')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

function evaluateAreaPassFail(areaAvgLux, pointValues, stdLuxStr, minLuxStr) {
  var isPass = true;
  stdLuxStr = (stdLuxStr || "").toString().trim();
  minLuxStr = (minLuxStr || "").toString().trim();

  if (stdLuxStr.indexOf(">") !== -1) {
    var threshold = parseFloat(stdLuxStr.replace(">", "")) || 2400;
    if (areaAvgLux < threshold) isPass = false;
  } else if (stdLuxStr.indexOf("-") !== -1 && !stdLuxStr.startsWith("-")) {
    var parts = stdLuxStr.split("-");
    var minVal = parseFloat(parts[0]) || 0;
    var maxVal = parseFloat(parts[1]) || 99999;
    if (areaAvgLux < minVal || areaAvgLux > maxVal) isPass = false;
  } else {
    var numStd = parseFloat(stdLuxStr) || 0;
    if (areaAvgLux < numStd) isPass = false;
  }

  if (minLuxStr !== "" && minLuxStr !== "-") {
    var numMin = parseFloat(minLuxStr);
    if (!isNaN(numMin) && numMin > 0) {
      for (var i = 0; i < pointValues.length; i++) {
        if (pointValues[i] < numMin) { isPass = false; break; }
      }
    }
  }
  return isPass ? "ผ่าน" : "ไม่ผ่าน";
}

// 2.1 บันทึกข้อมูลแบบพื้นที่ (Database_Area)
function saveAreaBatchData(records) {
  try {
    if (!records) throw new Error("ไม่พบชุดข้อมูลที่ส่งมา");
    if (!Array.isArray(records)) records = [records];
    if (records.length === 0) throw new Error("ชุดข้อมูลว่างเปล่า");

    var ss = SpreadsheetApp.getActiveSpreadsheet();
    if (!ss) throw new Error("ไม่สามารถเชื่อมต่อ Google Spreadsheet ได้");

    var sheet = ss.getSheetByName("Database_Area");
    if (!sheet) {
      sheet = ss.insertSheet("Database_Area");
      sheet.appendRow([
        "Timestamp", "วันที่ตรวจวัด", "เวลาที่เริ่มวัด", "สภาพอากาศ", 
        "เครื่องมือที่ใช้", "หมายเลขเครื่อง", "วันที่สอบเทียบล่าสุด", 
        "ส่วนงาน/คณะ", "ชื่ออาคาร", "เลขห้อง", 
        "รหัสจุดตรวจวัด", "ลักษณะพื้นที่การทำงาน", "รูปแบบการเก็บข้อมูล",
        "ค่าที่วัดได้รายจุด (Lux)", "ค่ามาตรฐาน (Lux)", "จุดที่ความเข้มแสงต่ำสุด (Lux)", 
        "ค่าเฉลี่ยรวมพื้นที่ (Lux)", "ผลการประเมิน", "หมายเหตุ", "ผู้ตรวจวัด (Email)"
      ]);
      sheet.getRange("A1:T1").setFontWeight("bold").setBackground("#d9ead3");
      sheet.setFrozenRows(1);
    }

    var mainData = [];
    var timestamp = new Date();
    var userEmail = Session.getActiveUser().getEmail() || "Field Inspector (Open Form)";

    for (var i = 0; i < records.length; i++) {
      var rec = records[i];
      var areaAvgLux = parseFloat(rec.measuredLux) || 0;
      var rawData = typeof rec.rawData === "string" ? JSON.parse(rec.rawData || "{}") : (rec.rawData || {});
      var pId = String(rec.patternId || "1");
      var subPoints = [];

      if (pId === "1" || pId === "8") {
        var count = parseInt(rawData.pointCount || "0");
        var prefix = (pId === "1") ? "r-" : "E-";
        for (var p = 1; p <= count; p++) {
          var v1 = parseFloat(rawData[prefix + p + '_1'] || 0);
          var v2 = parseFloat(rawData[prefix + p + '_2'] || 0);
          var v3 = parseFloat(rawData[prefix + p + '_3'] || 0);
          subPoints.push({ code: prefix + p, val: Number(((v1 + v2 + v3) / 3).toFixed(2)) });
        }
      } else {
        var ptList = [];
        if (pId === "2") ptList = ['P-1', 'P-2', 'Q-1', 'Q-2', 'Q-3', 'Q-4', 'R-1', 'R-2', 'R-3', 'R-4', 'R-5', 'R-6', 'R-7', 'R-8', 'T-1', 'T-2', 'T-3', 'T-4'];
        else if (pId === "3") ptList = ['P-1', 'P-2', 'P-3', 'P-4'];
        else if (pId === "4") ptList = ['q-1', 'q-2', 'q-3', 'q-4', 'q-5', 'q-6', 'q-7', 'q-8', 'p-1', 'p-2'];
        else if (pId === "5") ptList = ['r-1', 'r-2', 'r-3', 'r-4', 'q-1', 'q-2', 't-1', 't-2', 't-3', 't-4', 'p-1', 'p-2'];
        else if (pId === "6") ptList = ['q-1', 'q-2', 'q-3', 'q-4', 'q-5', 'q-6', 'p-1', 'p-2'];
        else if (pId === "7") ptList = ['r-1', 'r-2', 'r-3', 'r-4', 'q-1', 'q-2', 't-1', 't-2', 'p-1'];

        ptList.forEach(function(pt) {
          var v1 = parseFloat(rawData[pt + '_1'] || 0);
          var v2 = parseFloat(rawData[pt + '_2'] || 0);
          var v3 = parseFloat(rawData[pt + '_3'] || 0);
          subPoints.push({ code: pt, val: Number(((v1 + v2 + v3) / 3).toFixed(2)) });
        });
      }

      var ptVals = subPoints.map(function(sp) { return sp.val; });
      var evalStatus = evaluateAreaPassFail(areaAvgLux, ptVals, rec.standardLux, rec.minLux);

      subPoints.forEach(function(sp) {
        mainData.push([
          timestamp, rec.date, rec.time, rec.weather,
          rec.equipment, rec.serialNo, rec.calDate,
          rec.department, rec.building, rec.room,
          sp.code, rec.task, rec.pattern,
          sp.val, rec.standardLux, rec.minLux,
          areaAvgLux, evalStatus, rec.controlMeasure || "-", userEmail
        ]);
      });
    }

    if (mainData.length > 0) {
      sheet.getRange(sheet.getLastRow() + 1, 1, mainData.length, mainData[0].length).setValues(mainData);
    }
    return "บันทึกข้อมูลแบบพื้นที่สำเร็จ (" + mainData.length + " จุดตรวจวัด)";
  } catch (err) {
    throw new Error("ข้อผิดพลาดบน Server: " + err.message);
  }
}

function getExistingLocations() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Database_Area") || ss.getSheetByName("Database");
  if (!sheet || sheet.getLastRow() <= 1) return {};

  var data = sheet.getRange(2, 8, sheet.getLastRow() - 1, 3).getDisplayValues();
  var tree = {};

  for (var i = 0; i < data.length; i++) {
    var dept = (data[i][0] || "").trim();
    var bld = (data[i][1] || "").trim();
    var room = (data[i][2] || "").trim();

    if (!dept || !bld || !room) continue;

    if (!tree[dept]) tree[dept] = {};
    if (!tree[dept][bld]) tree[dept][bld] = [];
    if (tree[dept][bld].indexOf(room) === -1) {
      tree[dept][bld].push(room);
    }
  }
  return tree;
}

function getRoomDetailsForEdit(dept, bld, room) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Database_Area") || ss.getSheetByName("Database");
  if (!sheet || sheet.getLastRow() <= 1) return { error: "ไม่พบฐานข้อมูล" };

  var lastRow = sheet.getLastRow();
  var data = sheet.getRange(2, 1, lastRow - 1, sheet.getLastColumn()).getDisplayValues();
  var roomRows = [];

  for (var i = 0; i < data.length; i++) {
    if ((data[i][7] || "").trim() === dept &&
        (data[i][8] || "").trim() === bld &&
        (data[i][9] || "").trim() === room) {
      roomRows.push(data[i]);
    }
  }

  if (roomRows.length === 0) return { error: "ไม่พบข้อมูลห้องที่เลือก" };

  var first = roomRows[0];
  var pointsList = roomRows.map(function(r) {
    return {
      pointCode: r[10] || "-",
      pointLux: r[13] || r[14] || "0"
    };
  });

  return {
    date: first[1],
    time: first[2],
    weather: first[3],
    equipment: first[4],
    serialNo: first[5],
    calDate: first[6],
    department: first[7],
    building: first[8],
    room: first[9],
    task: first[11],
    patternText: first[12],
    standardLux: first[14] || first[15],
    minLux: first[15] || first[16] || "-",
    measuredLux: first[16] || first[17] || "0",
    evaluation: first[17] || first[18],
    recommendation: first[18] || first[19] || "-",
    pointsList: pointsList,
    fullRows: roomRows
  };
}

      function calcSpot(id) {
        const card = document.getElementById(`spotWorker_${id}`);
        const avg = (c1, c2, c3) => {
          const v1 = parseFloat(card.querySelector(c1).value) || 0;
          const v2 = parseFloat(card.querySelector(c2).value) || 0;
          const v3 = parseFloat(card.querySelector(c3).value) || 0;
          return ((v1 + v2 + v3) / 3).toFixed(2);
        };
        card.querySelector('.sw-avg1').innerText = avg('.sw-w1', '.sw-w2', '.sw-w3');
        card.querySelector('.sw-avg2').innerText = avg('.sw-s1-1', '.sw-s1-2', '.sw-s1-3');
        card.querySelector('.sw-avg3').innerText = avg('.sw-s2-1', '.sw-s2-2', '.sw-s2-3');
      }

            alert("บันทึกไม่สำเร็จ: " + err.message);
          })
